// ==========================================
// PRUEBA 7: Recursividad
// ==========================================
int factorial(int n) {
    if (n < 2) {
        return 1;
    }
    int anterior = n - 1;
    return n * factorial(anterior);
}

int resultado_fac = factorial(5);
print(resultado_fac);

// PRUEBA 11: Recursividad Múltiple (Fibonacci)
// Evalúa la pila de ejecución y múltiples retornos
// ==========================================
int fibonacci(int n) {
    if (n < 2) {
        return n;
    }
    int a = fibonacci(n - 1);
    int b = fibonacci(n - 2);
    return a + b;
}

int res_fib = fibonacci(7);
print(res_fib);

// PRUEBA 14: Anidación Profunda y Bloques Else
// ==========================================
int x = 0;
int y = 0;
int cont = 0;
while (x < 5) {
    y = 0;
    while (y < 5) {
        if (x == y) {
            cont = cont + 1;
        } else {
            if (x > y) {
                cont = cont + 2;
            }
        }
        y = y + 1;
    }
    x = x + 1;
}
print(cont);

// ==========================================
// PRUEBA 15: Algoritmo Real (Búsqueda Binaria)
// Integra arreglos, operaciones matemáticas condicionales y mutación de estado
// ==========================================
int datos[7];
datos[0] = 2; 
datos[1] = 5; 
datos[2] = 8; 
datos[3] = 12;
datos[4] = 16; 
datos[5] = 23; 
datos[6] = 38;

int objetivo = 23;
int inicio = 0;
int fin = 6;
int encontrado = 0; 
int medio = 0;

while (inicio < fin + 1) {
    medio = (inicio + fin) / 2;
    
    if (datos[medio] == objetivo) {
        encontrado = 1;
        inicio = fin + 1; 
    } else {
        if (datos[medio] < objetivo) {
            inicio = medio + 1;
        } else {
            fin = medio - 1;
        }
    }
}
print(encontrado);
