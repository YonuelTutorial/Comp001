// PRUEBA 02 - FACTORIAL
int factorialAvanzado(int n) {
    if (n <= 1) {
        return 1;
    }
    return n * factorialAvanzado(n - 1);
}
print(factorialAvanzado(6));

// PRUEBA 03 - FIBONACCI
int fibonacciAvanzado(int n) {
    if (n <= 1) {
        return n;
    }
    return fibonacciAvanzado(n - 1) + fibonacciAvanzado(n - 2);
}
print(fibonacciAvanzado(10));

// PRUEBA 04 - PRIMOS
bool esPrimoAvanzado(int n) {
    if (n < 2) {
        return false;
    }
    int divisor = 2;
    while (divisor * divisor <= n) {
        if ((n / divisor) * divisor == n) {
            return false;
        }
        divisor = divisor + 1;
    }
    return true;
}
print(esPrimoAvanzado(29));
print(esPrimoAvanzado(30));

// PRUEBA 05 - PERFECTOS
bool esPerfectoAvanzado(int n) {
    if (n < 2) {
        return false;
    }
    int divisor = 1;
    int suma = 0;
    while (divisor < n) {
        if ((n / divisor) * divisor == n) {
            suma = suma + divisor;
        }
        divisor = divisor + 1;
    }
    return suma == n;
}
print(esPerfectoAvanzado(6));
print(esPerfectoAvanzado(28));
print(esPerfectoAvanzado(12));

// PRUEBA 06 - CAPICUAS
int invertirNumero(int n) {
    int invertido = 0;
    int digito = 0;
    while (n > 0) {
        digito = n - (n / 10) * 10;
        invertido = invertido * 10 + digito;
        n = n / 10;
    }
    return invertido;
}
bool esCapicua(int n) {
    return n == invertirNumero(n);
}
print(esCapicua(12321));
print(esCapicua(12345));

// PRUEBA 07 - SUMA DE DIGITOS
int sumaDigitos(int n) {
    int suma = 0;
    while (n > 0) {
        suma = suma + n - (n / 10) * 10;
        n = n / 10;
    }
    return suma;
}
print(sumaDigitos(9876));

// PRUEBA 08 - POTENCIA RECURSIVA
int potenciaRecursiva(int base, int exponente) {
    if (exponente == 0) {
        return 1;
    }
    return base * potenciaRecursiva(base, exponente - 1);
}
print(potenciaRecursiva(2, 10));

// PRUEBA 09 - MAXIMO COMUN DIVISOR
int maximoComunDivisor(int a, int b) {
    int residuo = 0;
    while (b != 0) {
        residuo = a - (a / b) * b;
        a = b;
        b = residuo;
    }
    return a;
}
print(maximoComunDivisor(84, 30));

// PRUEBA 12 - RECURSIVIDAD MUTUA
bool numeroPar(int n) {
    if (n == 0) {
        return true;
    }
    return numeroImpar(n - 1);
}

    if (n == 0) {
        return false;
    }
    return numeroPar(n - 1);
}

// PRUEBA 13 - NUMERO TRIANGULAR
int numeroTriangular(int n) {
    if (n <= 1) {
        return n;
    }
    return n + numeroTriangular(n - 1);
}

// PRUEBA 18 - ANO BISIESTO
bool divisiblePor(int numero, int divisor) {
    return (numero / divisor) * divisor == numero;
}
bool esBisiesto(int ano) {
    if (divisiblePor(ano, 400)) {
        return true;
    }
    if (divisiblePor(ano, 100)) {
        return false;
    }
    return divisiblePor(ano, 4);
}
print(esBisiesto(2024));
print(esBisiesto(1900));
print(esBisiesto(2000));

// PRUEBA 20 - COLLATZ
int pasosCollatz(int n) {
    int pasos = 0;
    while (n != 1) {
        if ((n / 2) * 2 == n) {
            n = n / 2;
        } else {
            n = n * 3 + 1;
        }
        pasos = pasos + 1;
    }
    return pasos;
}
print(pasosCollatz(6));
